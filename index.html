    <script>

        var gridDisplay = document.getElementById('grid');

        var scoreDisplay = document.getElementById('score');

        var squares = [];

        var score = 0;


        var idolImages = {

            2: ['2.jpg', '2.JPG'],

            4: ['4.jpg', '4.JPG'],

            8: ['8.jpg', '8.JPG'],

            16: ['16.jpg', '16.JPG'],

            32: ['32.jpg', '32.JPG'],

            64: ['64.jpg', '64.JPG'],

            128: ['128.jpg', '128.JPG'],

            256: ['256.jpg', '256.JPG'],

            512: ['512.jpg', '512.JPG'],

            1024: ['1024.jpg', '1024.JPG'],

            2048: ['2048.jpg', '2048.JPG']

        };


        function createBoard() {

            gridDisplay.innerHTML = '';

            squares = [];

            for (var i = 0; i < 16; i++) {

                var square = document.createElement('div');

                square.classList.add('cell');

                gridDisplay.appendChild(square);

                squares.push(square);

            }

            generate();

            generate();

            updateBoard();

        }


        function generate() {

            var emptySquares = [];

            for (var i = 0; i < squares.length; i++) {

                if (!squares[i].getAttribute('data-value')) {

                    emptySquares.push(squares[i]);

                }

            }

            if (emptySquares.length > 0) {

                var randomSquare = emptySquares[Math.floor(Math.random() * emptySquares.length)];

                randomSquare.setAttribute('data-value', Math.random() > 0.1 ? 2 : 4);

                updateBoard();

            }

        }


        function updateBoard() {

            for (var i = 0; i < squares.length; i++) {

                var sq = squares[i];

                var val = sq.getAttribute('data-value');

                sq.innerHTML = '';

                if (val && val > 0) {

                    if (idolImages[val]) {

                        var img = document.createElement('img');

                        img.src = idolImages[val][0];

                        img.onerror = function() {

                            this.onerror = null;

                            this.src = this.src.endsWith('.jpg') ? this.src.replace('.jpg', '.JPG') : this.src.replace('.JPG', '.jpg');

                        };

                        sq.appendChild(img);

                    } else {

                        sq.innerText = val;

                    }

                }

            }

            scoreDisplay.innerHTML = score;

        }


        function slide(row) {

            var arr = [];

            for (var i = 0; i < row.length; i++) {

                if (row[i] !== 0) arr.push(row[i]);

            }

            while (arr.length < 4) {

                arr.push(0);

            }

            return arr;

        }


        function combine(row) {

            for (var i = 0; i < 3; i++) {

                if (row[i] === row[i + 1] && row[i] !== 0) {

                    row[i] *= 2;

                    score += row[i];

                    row[i + 1] = 0;

                }

            }

            return row;

        }


        function getValue(idx) {

            var val = squares[idx].getAttribute('data-value');

            return val ? parseInt(val) : 0;

        }


        function setValue(idx, val) {

            if (val && val > 0) {

                squares[idx].setAttribute('data-value', val);

            } else {

                squares[idx].removeAttribute('data-value');

            }

        }


        function moveLeft() {

            var changed = false;

            for (var i = 0; i < 16; i += 4) {

                var row = [getValue(i), getValue(i+1), getValue(i+2), getValue(i+3)];

                var original = row.slice();

                var res = slide(combine(slide(row)));

                for (var j = 0; j < 4; j++) {

                    setValue(i + j, res[j]);

                    if (res[j] !== original[j]) changed = true;

                }

            }

            if (changed) generate();

        }


        function moveRight() {

            var changed = false;

            for (var i = 0; i < 16; i += 4) {

                var row = [getValue(i), getValue(i+1), getValue(i+2), getValue(i+3)];

                var original = row.slice();

                var reversed = row.slice().reverse();

                var res = slide(combine(slide(reversed))).reverse();

                for (var j = 0; j < 4; j++) {

                    setValue(i + j, res[j]);

                    if (res[j] !== original[j]) changed = true;

                }

            }

            if (changed) generate();

        }


        function moveUp() {

            var changed = false;

            for (var i = 0; i < 4; i++) {

                var col = [getValue(i), getValue(i+4), getValue(i+8), getValue(i+12)];

                var original = col.slice();

                var res = slide(combine(slide(col)));

                for (var j = 0; j < 4; j++) {

                    setValue(i + j * 4, res[j]);

                    if (res[j] !== original[j]) changed = true;

                }

            }

            if (changed) generate();

        }


        function moveDown() {

            var changed = false;

            for (var i = 0; i < 4; i++) {

                var col = [getValue(i), getValue(i+4), getValue(i+8), getValue(i+12)];

                var original = col.slice();

                var reversed = col.slice().reverse();

                var res = slide(combine(slide(reversed))).reverse();

                for (var j = 0; j < 4; j++) {

                    setValue(i + j * 4, res[j]);

                    if (res[j] !== original[j]) changed = true;

                }

            }

            if (changed) generate();

        }


        // Bắt sự kiện bàn phím cho Máy tính

        document.addEventListener('keydown', function(e) {

            if (e.key === 'ArrowLeft') { e.preventDefault(); moveLeft(); }

            else if (e.key === 'ArrowRight') { e.preventDefault(); moveRight(); }

            else if (e.key === 'ArrowUp') { e.preventDefault(); moveUp(); }

            else if (e.key === 'ArrowDown') { e.preventDefault(); moveDown(); }

        });


        // Bắt sự kiện Vuốt cho Điện thoại

        var startX = 0, startY = 0;


        document.addEventListener('touchstart', function(e) {

            startX = e.touches[0].clientX;

            startY = e.touches[0].clientY;

        }, false);


        document.addEventListener('touchmove', function(e) {

            e.preventDefault();

        }, { passive: false });


        document.addEventListener('touchend', function(e) {

            var dx = e.changedTouches[0].clientX - startX;

            var dy = e.changedTouches[0].clientY - startY;

            if (Math.max(Math.abs(dx), Math.abs(dy)) > 20) {

                if (Math.abs(dx) > Math.abs(dy)) {

                    if (dx > 0) moveRight(); else moveLeft();

                } else {

                    if (dy > 0) moveDown(); else moveUp();

                }

            }

        }, false);


        createBoard();
