    <script>

        const gridDisplay = document.getElementById('grid');

        const scoreDisplay = document.getElementById('score');

        let squares = [];

        let score = 0;


        const idolImages = {

            2: ['2.jpg', '2.JPG'],

            4: ['4.jpg', '4.JPG'],

            8: ['8.jpg', '8.JPG'],

            16: ['16.jpg', '16.JPG'],

            32: ['32.jpg', '32.JPG'],

            64: ['64.jpg', '64.JPG'],

            128: ['128.jpg', '128.JPG'],

            256: ['256.jpg', '256.JPG'],

            512: ['512.jpg', '512.JPG'],

            1024: ['1024.jpg', '1024.JPG'],

            2048: ['2048.jpg', '2048.JPG']

        };


        function createBoard() {

            gridDisplay.innerHTML = '';

            squares = [];

            for (let i = 0; i < 16; i++) {

                let square = document.createElement('div');

                square.classList.add('cell');

                gridDisplay.appendChild(square);

                squares.push(square);

            }

            generate();

            generate();

            updateBoard();

        }


        function generate() {

            let emptySquares = squares.filter(sq => !sq.dataset.value);

            if (emptySquares.length > 0) {

                let randomSquare = emptySquares[Math.floor(Math.random() * emptySquares.length)];

                randomSquare.dataset.value = Math.random() > 0.1 ? 2 : 4;

                updateBoard();

            }

        }


        function updateBoard() {

            squares.forEach(sq => {

                let val = sq.dataset.value;

                sq.innerHTML = '';

                if (val) {

                    if (idolImages[val]) {

                        let img = document.createElement('img');

                        img.src = idolImages[val][0];

                        img.onerror = function() {

                            if (idolImages[val][1]) this.src = idolImages[val][1];

                            else { this.style.display = 'none'; sq.innerText = val; }

                        };

                        sq.appendChild(img);

                    } else {

                        sq.innerHTML = val;

                    }

                }

            });

            scoreDisplay.innerHTML = score;

        }


        function slide(row) {

            let arr = row.filter(val => val);

            let missing = 4 - arr.length;

            return arr.concat(Array(missing).fill(0));

        }


        function combine(row) {

            for (let i = 0; i < 3; i++) {

                if (row[i] === row[i + 1] && row[i] !== 0) {

                    row[i] *= 2;

                    score += row[i];

                    row[i + 1] = 0;

                }

            }

            return row;

        }


        function moveLeft() {

            let changed = false;

            for (let i = 0; i < 16; i += 4) {

                let row = [parseInt(squares[i].dataset.value)||0, parseInt(squares[i+1].dataset.value)||0, parseInt(squares[i+2].dataset.value)||0, parseInt(squares[i+3].dataset.value)||0];

                let original = [...row];

                let res = slide(combine(slide(row)));

                for (let j = 0; j < 4; j++) {

                    if (res[j] !== 0) squares[i+j].dataset.value = res[j];

                    else delete squares[i+j].dataset.value;

                    if (res[j] !== original[j]) changed = true;

                }

            }

            if (changed) generate();

        }


        function moveRight() {

            let changed = false;

            for (let i = 0; i < 16; i += 4) {

                let row = [parseInt(squares[i].dataset.value)||0, parseInt(squares[i+1].dataset.value)||0, parseInt(squares[i+2].dataset.value)||0, parseInt(squares[i+3].dataset.value)||0];

                let original = [...row];

                let res = slide(combine(slide([...row].reverse()))).reverse();

                for (let j = 0; j < 4; j++) {

                    if (res[j] !== 0) squares[i+j].dataset.value = res[j];

                    else delete squares[i+j].dataset.value;

                    if (res[j] !== original[j]) changed = true;

                }

            }

            if (changed) generate();

        }


        function moveUp() {

            let changed = false;

            for (let i = 0; i < 4; i++) {

                let col = [parseInt(squares[i].dataset.value)||0, parseInt(squares[i+4].dataset.value)||0, parseInt(squares[i+8].dataset.value)||0, parseInt(squares[i+12].dataset.value)||0];

                let original = [...col];

                let res = slide(combine(slide(col)));

                for (let j = 0; j < 4; j++) {

                    let idx = i + j * 4;

                    if (res[j] !== 0) squares[idx].dataset.value = res[j];

                    else delete squares[idx].dataset.value;

                    if (res[j] !== original[j]) changed = true;

                }

            }

            if (changed) generate();

        }


        function moveDown() {

            let changed = false;

            for (let i = 0; i < 4; i++) {

                let col = [parseInt(squares[i].dataset.value)||0, parseInt(squares[i+4].dataset.value)||0, parseInt(squares[i+8].dataset.value)||0, parseInt(squares[i+12].dataset.value)||0];

                let original = [...col];

                let res = slide(combine(slide([...col].reverse()))).reverse();

                for (let j = 0; j < 4; j++) {

                    let idx = i + j * 4;

                    if (res[j] !== 0) squares[idx].dataset.value = res[j];

                    else delete squares[idx].dataset.value;

                    if (res[j] !== original[j]) changed = true;

                }

            }

            if (changed) generate();

        }


        // Bắt sự kiện vuốt cảm ứng trên điện thoại

        let startX = 0, startY = 0;

        document.addEventListener('touchstart', e => {

            startX = e.touches[0].clientX;

            startY = e.touches[0].clientY;

        }, {passive: false});


        document.addEventListener('touchmove', e => {

            e.preventDefault(); // Chặn đứng hoàn toàn việc cuộn trang

        }, {passive: false});


        document.addEventListener('touchend', e => {

            let dx = e.changedTouches[0].clientX - startX;

            let dy = e.changedTouches[0].clientY - startY;

            if (Math.max(Math.abs(dx), Math.abs(dy)) > 25) {

                if (Math.abs(dx) > Math.abs(dy)) {

                    if (dx > 0) moveRight(); else moveLeft();

                } else {

                    if (dy > 0) moveDown(); else moveUp();

                }

            }

        }, {passive: false});


        createBoard();
